1️⃣ Encryption Utility (AES-256-GCM)
/services/kycVault.service.js
const crypto = require("crypto");

const MASTER_KEY = Buffer.from(process.env.KYC_MASTER_KEY, "hex"); // 32 bytes

function encrypt(text) {
  const iv = crypto.randomBytes(12);
  const cipher = crypto.createCipheriv("aes-256-gcm", MASTER_KEY, iv);

  let encrypted = cipher.update(text, "utf8", "hex");
  encrypted += cipher.final("hex");

  const authTag = cipher.getAuthTag().toString("hex");

  return {
    iv: iv.toString("hex"),
    content: encrypted,
    tag: authTag
  };
}

function decrypt(payload) {
  const decipher = crypto.createDecipheriv(
    "aes-256-gcm",
    MASTER_KEY,
    Buffer.from(payload.iv, "hex")
  );

  decipher.setAuthTag(Buffer.from(payload.tag, "hex"));

  let decrypted = decipher.update(payload.content, "hex", "utf8");
  decrypted += decipher.final("utf8");

  return decrypted;
}

module.exports = { encrypt, decrypt };
✔ Unique IV
✔ Auth tag integrity
✔ No plaintext stored
2️⃣ Audit Hash Chain
/services/audit.service.js
const crypto = require("crypto");
const pool = require("../db");

async function appendAudit(client, kycProfileId, eventType, payload) {
  const previous = await client.query(
    `SELECT block_hash FROM kyc_audit_log 
     WHERE kyc_profile_id=$1 
     ORDER BY timestamp DESC LIMIT 1`,
    [kycProfileId]
  );

  const previousHash = previous.rows[0]?.block_hash || null;

  const payloadHash = crypto
    .createHash("sha256")
    .update(JSON.stringify(payload))
    .digest("hex");

  const blockHash = crypto
    .createHash("sha256")
    .update(payloadHash + (previousHash || ""))
    .digest("hex");

  await client.query(
    `INSERT INTO kyc_audit_log
     (kyc_profile_id, event_type, event_payload_hash, previous_block_hash, block_hash, timestamp)
     VALUES ($1,$2,$3,$4,$5,NOW())`,
    [kycProfileId, eventType, payloadHash, previousHash, blockHash]
  );
}

module.exports = { appendAudit };
Tamper-resistant.
3️⃣ KYC State Machine
/services/kycStateMachine.service.js
const allowedTransitions = {
  DRAFT: ["PLATFORM_VERIFIED"],
  PLATFORM_VERIFIED: ["REGULATORY_PENDING"],
  REGULATORY_PENDING: ["REGULATORY_VERIFIED", "REJECTED"],
  REGULATORY_VERIFIED: ["EXPIRED"]
};

function validateTransition(current, next) {
  if (!allowedTransitions[current]?.includes(next)) {
    throw new Error(`Invalid KYC transition ${current} → ${next}`);
  }
}

module.exports = { validateTransition };
No illegal transitions.
4️⃣ KYC Orchestrator (Atomic Write-Back)
/services/kycOrchestrator.service.js
const pool = require("../db");
const { encrypt } = require("./kycVault.service");
const { validateTransition } = require("./kycStateMachine.service");
const { appendAudit } = require("./audit.service");

async function persistPANAndAadhaar(userId, panData, aadhaarData) {
  const client = await pool.connect();

  try {
    await client.query("BEGIN");

    const profileRes = await client.query(
      `SELECT id, kyc_status FROM kyc_profiles WHERE user_id=$1 FOR UPDATE`,
      [userId]
    );

    const profile = profileRes.rows[0];
    if (!profile) throw new Error("KYC profile not found");

    const kycProfileId = profile.id;

    // Encrypt data
    const encryptedPAN = encrypt(panData.pan);
    const encryptedName = encrypt(panData.name);
    const encryptedDOB = encrypt(aadhaarData.dob);
    const encryptedAddress = encrypt(aadhaarData.address);

    // Upsert vault
    await client.query(
      `INSERT INTO kyc_vault
       (kyc_profile_id, pan_encrypted, full_name_encrypted,
        dob_encrypted, address_line1_encrypted,
        aadhaar_last_four, created_at)
       VALUES ($1,$2,$3,$4,$5,$6,NOW())
       ON CONFLICT (kyc_profile_id)
       DO UPDATE SET
         pan_encrypted=$2,
         full_name_encrypted=$3,
         dob_encrypted=$4,
         address_line1_encrypted=$5,
         aadhaar_last_four=$6,
         updated_at=NOW()`,
      [
        kycProfileId,
        JSON.stringify(encryptedPAN),
        JSON.stringify(encryptedName),
        JSON.stringify(encryptedDOB),
        JSON.stringify(encryptedAddress),
        aadhaarData.lastFour
      ]
    );

    validateTransition(profile.kyc_status, "PLATFORM_VERIFIED");

    await client.query(
      `UPDATE kyc_profiles
       SET kyc_status='PLATFORM_VERIFIED', updated_at=NOW()
       WHERE id=$1`,
      [kycProfileId]
    );

    await appendAudit(client, kycProfileId, "PLATFORM_VERIFIED", {
      panVerified: true,
      aadhaarVerified: true
    });

    await client.query("COMMIT");
  } catch (err) {
    await client.query("ROLLBACK");
    throw err;
  } finally {
    client.release();
  }
}

module.exports = { persistPANAndAadhaar };

✔ Locked row
✔ Encrypted
✔ Audit appended

5️⃣ Regulatory Verification Persistence
async function markRegulatoryVerified(userId, regulatoryData) {
  const client = await pool.connect();
  try {
    await client.query("BEGIN");

    const profile = await client.query(
      `SELECT id, kyc_status FROM kyc_profiles WHERE user_id=$1 FOR UPDATE`,
      [userId]
    );

    const kycProfileId = profile.rows[0].id;

    validateTransition(profile.rows[0].kyc_status, "REGULATORY_VERIFIED");

    await client.query(
      `INSERT INTO kyc_regulatory
       (kyc_profile_id, kra_agency, kra_status,
        ckyc_number, verification_source,
        verification_reference_id, verification_timestamp)
       VALUES ($1,$2,'VERIFIED',$3,$4,$5,NOW())`,
      [
        kycProfileId,
        regulatoryData.kraAgency,
        regulatoryData.ckyc,
        regulatoryData.source,
        regulatoryData.refId
      ]
    );

    await client.query(
      `UPDATE kyc_profiles
       SET kyc_status='REGULATORY_VERIFIED',
           locked_at=NOW()
       WHERE id=$1`,
      [kycProfileId]
    );

    await appendAudit(client, kycProfileId, "REGULATORY_VERIFIED", regulatoryData);

    await client.query("COMMIT");
  } catch (e) {
    await client.query("ROLLBACK");
    throw e;
  } finally {
    client.release();
  }
}

module.exports = async function (req, res, next) {
  const { userId } = req.user;

  const result = await pool.query(
    `SELECT kyc_status FROM kyc_profiles WHERE user_id=$1`,
    [userId]
  );

  if (result.rows[0].kyc_status !== "REGULATORY_VERIFIED") {
    return res.status(403).json({ error: "KYC_NOT_VERIFIED" });
  }

  next();
};

async function getPrefilledKYC(userId) {
  const result = await pool.query(
    `SELECT pan_encrypted, full_name_encrypted,
            dob_encrypted, address_line1_encrypted,
            aadhaar_last_four
     FROM kyc_vault v
     JOIN kyc_profiles p ON v.kyc_profile_id=p.id
     WHERE p.user_id=$1`,
    [userId]
  );

  const vault = result.rows[0];

  return {
    pan: decrypt(JSON.parse(vault.pan_encrypted)),
    name: decrypt(JSON.parse(vault.full_name_encrypted)),
    dob: decrypt(JSON.parse(vault.dob_encrypted)),
    address: decrypt(JSON.parse(vault.address_line1_encrypted)),
    aadhaarLastFour: vault.aadhaar_last_four
  };
}
