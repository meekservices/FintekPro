const handleExportPNG = async () => {
  const element = document.getElementById('greeting-card');
  if (!element) return;

  // We use scale: 3 and window.devicePixelRatio to ensure the export 
  // matches the high-quality screen preview exactly.
  const canvas = await html2canvas(element, {
    useCORS: true,
    scale: 3, 
    logging: false,
    backgroundColor: null,
    onclone: (clonedDoc) => {
      const textWrapper = clonedDoc.getElementById('agent-details-text-wrapper');
      
      if (textWrapper) {
        // FORCE FLEX COLLAPSE PREVENTION
        textWrapper.style.display = 'flex';
        textWrapper.style.flexDirection = 'column';
        textWrapper.style.justifyContent = 'center';
        
        // Target each row explicitly to force vertical separation
        const rows = Array.from(textWrapper.children) as HTMLElement[];
        
        rows.forEach((row, index) => {
          row.style.display = 'block';
          row.style.position = 'relative';
          row.style.top = '0px'; // Clear any old absolute offsets
          
          // ADD FORCED PHYSICAL GAP (This prevents the overlap you see)
          row.style.marginBottom = '12px'; 
          
          // Force explicit line heights so fonts don't "bleed" into each other
          if (index === 0) { // Name
             row.style.fontSize = '22px';
             row.style.lineHeight = '28px';
             row.style.fontWeight = 'bold';
          } else if (index === 1) { // Designation
             row.style.fontSize = '16px';
             row.style.lineHeight = '20px';
             row.style.color = '#FFD700'; // Ensure Gold is forced
          } else { // Contact Info
             row.style.fontSize = '14px';
             row.style.lineHeight = '18px';
          }
        });
      }
    }
  });

  const link = document.createElement('a');
  link.download = `diwali-greeting.png`;
  link.href = canvas.toDataURL('image/png', 1.0);
  link.click();
};

Why this finally works:
The "Last Line" Success: The phone number was visible because there was nothing below it to "push" it. By adding marginBottom to the Name and Designation in the onclone step, we create that same "buffer" for the top lines.

Explicit Line Heights: Setting line-height: 28px for a 22px font guarantees 6 pixels of "empty air" that html2canvas cannot fill with other text.

Scale 3: This forces the browser to render the text at a much higher resolution before snapping the "photo," preventing the font from getting chunky and overlapping.
