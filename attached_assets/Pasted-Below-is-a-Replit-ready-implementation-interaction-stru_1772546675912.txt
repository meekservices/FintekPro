// src/modules/picks/scorer/UnlistedScorer.ts

export enum ListingStage {
  UNLISTED = 'unlisted',
  PRE_IPO = 'pre_ipo',
  GROWTH = 'growth',
  MATURE = 'mature'
}

export enum PricingStatus {
  ACTIVE = 'active',
  PUBLISHED = 'published',
  INACTIVE = 'inactive'
}

export interface Company {
  id: string
  name: string
  listing_stage: ListingStage
  pricing_status: PricingStatus
  published_buy_price?: number | null
  sector?: string
  volatility_score?: number
  governance_score?: number
}

export interface ScoreBreakdown {
  listingStageScore: number
  pricingScore: number
  sectorScore: number
  riskAdjustment: number
  governanceScore: number
  totalScore: number
}

export class UnlistedScorer {

  static VERSION = "2.0.0"
  static MIN_THRESHOLD = 15

  static score(company: Company): ScoreBreakdown {

    if (!company) {
      throw new Error("Invalid company object")
    }

    let listingStageScore = 0
    let pricingScore = 0
    let sectorScore = 0
    let governanceScore = 0
    let riskAdjustment = 0

    // Listing Stage Scoring
    switch (company.listing_stage) {
      case ListingStage.UNLISTED:
        listingStageScore = 10
        break
      case ListingStage.PRE_IPO:
        listingStageScore = 8
        break
      case ListingStage.GROWTH:
        listingStageScore = 6
        break
      case ListingStage.MATURE:
        listingStageScore = 4
        break
      default:
        throw new Error(`Unknown listing stage: ${company.listing_stage}`)
    }

    // Pricing Validation
    if (
      (company.pricing_status === PricingStatus.ACTIVE ||
        company.pricing_status === PricingStatus.PUBLISHED) &&
      company.published_buy_price &&
      company.published_buy_price > 0
    ) {
      pricingScore = 5
    }

    // Sector Bias (configurable)
    const highConvictionSectors = ['fintech', 'deeptech', 'ev', 'ai']
    if (company.sector && highConvictionSectors.includes(company.sector.toLowerCase())) {
      sectorScore = 7
    }

    // Governance
    governanceScore = company.governance_score ?? 0

    // Risk Adjustment (inverse volatility)
    const volatility = company.volatility_score ?? 5
    riskAdjustment = Math.max(0, 10 - volatility)

    const totalScore =
      listingStageScore +
      pricingScore +
      sectorScore +
      governanceScore +
      riskAdjustment

    return {
      listingStageScore,
      pricingScore,
      sectorScore,
      governanceScore,
      riskAdjustment,
      totalScore
    }
  }
}
🏗 Updated generateUnlistedPick()
export async function generateUnlistedPick() {

  const candidates = await getUnlistedCompanies()

  if (!candidates.length) {
    throw new Error("No unlisted companies available")
  }

  const scored = candidates.map(company => {
    const breakdown = UnlistedScorer.score(company)
    return { company, breakdown }
  })

  const eligible = scored.filter(s => 
    s.breakdown.totalScore >= UnlistedScorer.MIN_THRESHOLD
  )

  if (!eligible.length) {
    throw new Error("No companies met scoring threshold")
  }

  eligible.sort((a, b) => 
    b.breakdown.totalScore - a.breakdown.totalScore
  )

  const winner = eligible[0]

  await savePick({
    company_id: winner.company.id,
    category: 'unlisted',
    scoring_version: UnlistedScorer.VERSION,
    scoring_breakdown: winner.breakdown,
    risk_score: winner.breakdown.riskAdjustment
  })

  return winner
}
