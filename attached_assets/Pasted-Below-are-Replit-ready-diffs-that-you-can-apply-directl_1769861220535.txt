private parseSchemeTransactions(
  schemeText: string,
  meta: { isin: string; schemeName: string; amc: string; folio: string }
): PortfolioLot[] {
  const lots: PortfolioLot[] = [];

  const TXN_REGEX =
    /^(\d{2}-[A-Za-z]{3}-\d{4})\s+
     (Purchase|Purchase-BSE|SIP Purchase|Systematic Investment Purchase|Switch In)
     [^\d]*
     ([\d,]+\.\d{2})\s+
     ([\d,]+\.\d+)\s+
     ([\d.]+)/gmx;

  let match;
  while ((match = TXN_REGEX.exec(schemeText)) !== null) {
    lots.push({
      isin: meta.isin,
      schemeName: meta.schemeName,
      amc: meta.amc,
      folio: meta.folio,

      transactionDate: this.parseDate(match[1]),
      transactionType: this.normalizeTxn(match[2]),
      amount: this.parseNumber(match[3]),
      units: this.parseNumber(match[4]),
      nav: this.parseNumber(match[5]),

      source: 'CAS',
    });
  }

  return lots;
}

export function deriveHoldingsFromLots(lots: PortfolioLot[]) {
  const map = new Map<string, any>();

  for (const lot of lots) {
    const key = `${lot.isin}::${lot.folio}`;

    if (!map.has(key)) {
      map.set(key, {
        isin: lot.isin,
        schemeName: lot.schemeName,
        amc: lot.amc,
        folio: lot.folio,
        lots: [],
      });
    }

    map.get(key).lots.push(lot);
  }

  return Array.from(map.values()).map(h => ({
    ...h,
    totalUnits: h.lots.reduce((s, l) => s + l.units, 0),
    avgPrice:
      h.lots.reduce((s, l) => s + l.amount, 0) /
      h.lots.reduce((s, l) => s + l.units, 0),
  }));
}
