EPIC 1: Research List Core (Backend)
Ticket 1.1 – DB Schema: Research Lists
Type: Backend
Priority: P0
Task
Add advisor research list capability without breaking existing data.
Tables
CREATE TABLE research_lists (
  id UUID PRIMARY KEY,
  name TEXT NOT NULL,
  universe_type TEXT CHECK (universe_type IN ('MF','ETF','STOCK','BOND','FD')),
  created_by_agent_id UUID NOT NULL,
  visibility TEXT CHECK (visibility IN ('private','team','org')) DEFAULT 'private',
  is_editable BOOLEAN DEFAULT true,
  created_at TIMESTAMP DEFAULT now(),
  updated_at TIMESTAMP DEFAULT now()
);
Ticket 1.2 – DB Schema: Research List Items
Type: Backend
Priority: P0
CREATE TABLE research_list_items (
  id UUID PRIMARY KEY,
  research_list_id UUID REFERENCES research_lists(id) ON DELETE CASCADE,
  instrument_id UUID NOT NULL,
  instrument_type TEXT,
  added_source TEXT CHECK (added_source IN ('manual','screener','import')),
  added_at TIMESTAMP DEFAULT now()
);
