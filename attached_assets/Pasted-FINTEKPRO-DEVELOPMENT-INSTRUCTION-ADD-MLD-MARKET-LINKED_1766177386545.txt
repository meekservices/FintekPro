5. DATABASE SCHEMA — MLD MASTER TABLES
5.1 mld_master
mld_master (
  mld_id UUID PRIMARY KEY,
  isin TEXT UNIQUE,
  name TEXT,
  issuer TEXT,
  underlying TEXT,
  payoff_type TEXT,
  barrier_level DECIMAL,
  participation_rate DECIMAL,
  maturity_date DATE,
  issue_date DATE,
  face_value DECIMAL,
  min_investment DECIMAL,
  rating TEXT,
  risk_score INT,
  status ENUM('active','closed','matured','called_back'),
  liquidity_profile TEXT,
  is_listed BOOLEAN,
  is_published BOOLEAN DEFAULT FALSE,
  term_sheet_path TEXT,
  metadata JSONB,
  created_at TIMESTAMP,
  updated_at TIMESTAMP
);
