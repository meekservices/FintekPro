When html2canvas renders the PNG, it often calculates line heights differently than the browser preview, causing the "squashed" effect seen in your images.The following logic forces a clean, professional layout by utilizing a Flexbox approach during the cloning process.The Full onclone Logic for ReplitTell the Replit Agent to replace the current onclone function in your export component (likely where html2canvas is called) with this:TypeScriptconst handleExport = async () => {
  const element = document.getElementById('greeting-card');
  
  const canvas = await html2canvas(element, {
    onclone: (clonedDoc) => {
      const footerTextContainer = clonedDoc.getElementById('agent-details-container');
      if (footerTextContainer) {
        // 1. Clear any existing absolute positioning styles
        footerTextContainer.style.display = 'flex';
        footerTextContainer.style.flexDirection = 'column';
        
        // 2. Set a strict gap between rows (Solves the overlapping issue)
        footerTextContainer.style.gap = '12px'; 
        footerTextContainer.style.paddingLeft = '20px';
        footerTextContainer.style.justifyContent = 'center';

        // 3. Target specific lines for styling hierarchy
        const name = footerTextContainer.querySelector('.agent-name');
        const designation = footerTextContainer.querySelector('.agent-designation');
        const contacts = footerTextContainer.querySelectorAll('.agent-contact');

        if (name) {
          name.style.fontSize = '24px';
          name.style.fontWeight = 'bold';
          name.style.lineHeight = '1.2';
          name.style.marginBottom = '0px';
        }

        if (designation) {
          designation.style.fontSize = '18px';
          designation.style.color = '#FFD700'; // Gold
          designation.style.fontWeight = '500';
        }

        contacts.forEach(contact => {
          contact.style.fontSize = '16px';
          contact.style.display = 'flex';
          contact.style.alignItems = 'center';
          contact.style.gap = '8px';
        });
      }
    }
  });
  // ... rest of export logic
};
