const TRANSACTION_REGEX =
/(\d{2}-[A-Za-z]{3}-\d{4})\s+
(Purchase|Purchase-BSE|SIP Purchase|Systematic Investment Purchase|Switch In)
.*?
([\d,]+\.\d{2})\s+
([\d,]+\.\d{3})\s+
([\d,]+\.\d{3,4})/;
For each scheme + folio:
const transactions = [];

for (const line of schemeLines) {
  if (line.includes('***')) continue; // ignore metadata

  const match = line.match(TRANSACTION_REGEX);
  if (match) {
    transactions.push({
      date: parseDate(match[1]),
      type: normalizeType(match[2]),
      amount: parseNumber(match[3]),
      units: parseNumber(match[4]),
      nav: parseNumber(match[5]),
    });
  }
}
