import { PortfolioLot } from '../models/portfolioLot';

private parseSchemeTransactions(
  schemeText: string,
  meta: { isin: string; schemeName: string; amc: string; folio: string }
): PortfolioLot[] {
  const lots: PortfolioLot[] = [];

  // Matches CAMS/KFintech rows like:
  // 07-Oct-2024 Purchase-BSE ... 199,990.00 4,974.876 40.200 4,974.876
  const TXN_REGEX =
    /^(\d{2}-[A-Za-z]{3}-\d{4})\s+
      (Purchase|Purchase-BSE|SIP Purchase|Systematic Investment Purchase|Switch In|Equity Switch In)
      [^\d]*
      ([\d,]+\.\d{2})\s+
      ([\d,]+\.\d+)\s+
      ([\d.]+)/gmx;

  let m: RegExpExecArray | null;
  while ((m = TXN_REGEX.exec(schemeText)) !== null) {
    lots.push({
      isin: meta.isin,
      schemeName: meta.schemeName,
      amc: meta.amc,
      folio: meta.folio,
      transactionDate: this.parseDate(m[1]).toISOString().slice(0, 10),
      transactionType: this.normalizeTxn(m[2]),
      amount: this.parseNumber(m[3]),
      units: this.parseNumber(m[4]),
      nav: this.parseNumber(m[5]),
      source: 'CAS',
    });
  }

  return lots;
}

DIFF 3 — Derive holdings FROM LOTS (kill ISIN grouping)

Replace entire file contents with:
import { PortfolioLot } from '../models/portfolioLot';

export function deriveHoldingsFromLots(lots: PortfolioLot[]) {
  const map = new Map<string, any>();

  for (const lot of lots) {
    const key = `${lot.isin}::${lot.folio}`;
    if (!map.has(key)) {
      map.set(key, {
        isin: lot.isin,
        schemeName: lot.schemeName,
        amc: lot.amc,
        folio: lot.folio,
        lots: [] as PortfolioLot[],
      });
    }
    map.get(key).lots.push(lot);
  }

  return Array.from(map.values()).map(h => {
    const totalUnits = h.lots.reduce((s, l) => s + l.units, 0);
    const invested = h.lots.reduce((s, l) => s + l.amount, 0);
    return {
      ...h,
      totalUnits,
      avgPrice: invested / (totalUnits || 1),
    };
  });
}
