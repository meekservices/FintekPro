✅ TASK 2 — Service: Daily Price Updater
Create service file:
/services/dailyPriceUpdater.ts
import { getActiveInstruments, updateLastDailyUpdate } from "../repositories/instrumentRepo";
import { insertPriceIfNotExists } from "../repositories/priceRepo";
import { fetchLatestPrice } from "../providers/marketDataProvider";
import { logJob } from "../repositories/jobLogRepo";

export async function runDailyPriceUpdate() {
  const instruments = await getActiveInstruments();

  const yesterday = new Date();
  yesterday.setDate(yesterday.getDate() - 1);

  for (const instrument of instruments) {
    if (instrument.last_daily_update >= yesterday) continue;

    try {
      const price = await fetchLatestPrice(instrument);

      await insertPriceIfNotExists(
        instrument.id,
        yesterday,
        price
      );

      await updateLastDailyUpdate(instrument.id, yesterday);

      await logJob("DAILY_UPDATE", instrument.id, "SUCCESS", null);
    } catch (err: any) {
      await logJob("DAILY_UPDATE", instrument.id, "FAILED", err.message);
    }
  }
}

✅ TASK 3 — Service: Historical Backfill
Create:
/services/historicalBackfillService.ts
import { getIncompleteHistoricalInstruments, markHistoricalComplete } from "../repositories/instrumentRepo";
import { insertBulkPrices } from "../repositories/priceRepo";
import { fetchHistoricalPrices } from "../providers/marketDataProvider";
import { logJob } from "../repositories/jobLogRepo";

export async function runHistoricalBackfill() {
  const instruments = await getIncompleteHistoricalInstruments();

  for (const instrument of instruments) {
    try {
      const history = await fetchHistoricalPrices(instrument);

      await insertBulkPrices(instrument.id, history);

      await markHistoricalComplete(instrument.id);

      await logJob("HISTORICAL_BACKFILL", instrument.id, "SUCCESS", null);
    } catch (err: any) {
      await logJob("HISTORICAL_BACKFILL", instrument.id, "FAILED", err.message);
    }
  }
}

import cron from "node-cron";
import { runDailyPriceUpdate } from "../services/dailyPriceUpdater";
import { runHistoricalBackfill } from "../services/historicalBackfillService";

// 8:45 PM IST - Daily NAV update
cron.schedule("15 15 * * *", async () => {
  console.log("Running Daily Price Update...");
  await runDailyPriceUpdate();
}, {
  timezone: "Asia/Kolkata"
});

// 2:00 AM IST - Historical Backfill
cron.schedule("30 20 * * *", async () => {
  console.log("Running Historical Backfill...");
  await runHistoricalBackfill();
}, {
  timezone: "Asia/Kolkata"
});
Note:
Replit runs in UTC.
8:45 PM IST = 15:15 UTC
2:00 AM IST = 20:30 UTC (previous day UTC offset adjusted)

✅ TASK 7 — Safe Insert Logic
In price repository:
export async function insertPriceIfNotExists(
  instrumentId: number,
  date: Date,
  price: number
) {
  await db.query(
    `
    INSERT INTO instrument_prices (instrument_id, price_date, close_price)
    VALUES ($1, $2, $3)
    ON CONFLICT (instrument_id, price_date) DO NOTHING
    `,
    [instrumentId, date, price]
  );
}
