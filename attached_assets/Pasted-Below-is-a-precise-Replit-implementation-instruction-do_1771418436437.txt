Implementation Skeleton
import { solveQP } from "quadprog"
import { identity } from "mathjs"

export function optimizeTransitionPortfolio(
   mu,
   sigma,
   w0,
   categoryMap,
   sectorMap,
   config
) {

   const lambda = config.lambda
   const gamma = config.gamma

   const n = mu.length

   const I = identity(n)._data

   const H = math.add(
      math.multiply(2 * lambda, sigma),
      math.multiply(2 * gamma, I)
   )

   const f = math.multiply(-1,
      math.add(mu, math.multiply(2 * gamma, w0))
   )

   const constraints = buildConstraints(
      n,
      categoryMap,
      sectorMap,
      config,
      w0
   )

   const result = solveQP(H, f, constraints.A, constraints.b)

   return applyMinWeightFilter(result.solution, config)
}

6️⃣ Apply Minimum Weight Filter
After solving:
function applyMinWeightFilter(weights, config) {

   let filtered = weights.map(w =>
      w < config.minWeight ? 0 : w
   )

   const total = sum(filtered)

   return filtered.map(w => w / total)
}
This removes ₹110 artifacts.
