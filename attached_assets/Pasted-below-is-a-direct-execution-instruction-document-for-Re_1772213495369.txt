2️⃣ DATABASE INSTALLATION (PostgreSQL)
🔹 Migration 1: unlisted_equity_master
CREATE TABLE unlisted_equity_master (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    company_name VARCHAR(255) NOT NULL,
    cin VARCHAR(21) UNIQUE,
    pan VARCHAR(20),
    isin VARCHAR(20),
    sector VARCHAR(100),
    incorporation_date DATE,
    face_value NUMERIC,
    authorized_capital NUMERIC,
    paid_up_capital NUMERIC,
    liquidity_type VARCHAR(50) DEFAULT 'illiquid',
    compliance_status VARCHAR(50) DEFAULT 'pending',
    status VARCHAR(50) DEFAULT 'active',
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_unlisted_company_name ON unlisted_equity_master(company_name);
CREATE INDEX idx_unlisted_sector ON unlisted_equity_master(sector);
🔹 Migration 2: valuation history (append-only)
CREATE TABLE unlisted_equity_valuation_history (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    unlisted_equity_id UUID REFERENCES unlisted_equity_master(id),
    valuation_method VARCHAR(50) NOT NULL,
    price NUMERIC NOT NULL,
    valuation_date DATE NOT NULL,
    supporting_document_url TEXT,
    created_at TIMESTAMP DEFAULT NOW()
);

CREATE INDEX idx_unlisted_val_date ON unlisted_equity_valuation_history(valuation_date);
🔹 Migration 3: compliance flags
ALTER TABLE unlisted_equity_master
ADD COLUMN risk_category VARCHAR(50) DEFAULT 'very_high',
ADD COLUMN rebalance_eligible BOOLEAN DEFAULT FALSE,
ADD COLUMN liquidity_weight NUMERIC DEFAULT 0;
