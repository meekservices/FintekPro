🔐 Registration Challenge Endpoint
import { generateRegistrationOptions } from "@simplewebauthn/server";

app.post("/auth/webauthn/register/options", async (req, res) => {
  const user = req.user;

  const options = generateRegistrationOptions({
    rpName: "FintekPro",
    rpID: "fintekpro.com",
    userID: user.id,
    userName: user.email,
    timeout: 60000,
    attestationType: "none",
    authenticatorSelection: {
      authenticatorAttachment: "platform",
      userVerification: "required"
    }
  });

  await redis.set(`challenge:${user.id}`, options.challenge, "EX", 300);

  res.json(options);
});
🔍 Verify Registration
import { verifyRegistrationResponse } from "@simplewebauthn/server";

app.post("/auth/webauthn/register/verify", async (req, res) => {
  const expectedChallenge = await redis.get(`challenge:${req.user.id}`);

  const verification = await verifyRegistrationResponse({
    response: req.body,
    expectedChallenge,
    expectedOrigin: "https://fintekpro.com",
    expectedRPID: "fintekpro.com"
  });

  if (verification.verified) {
    const { credentialPublicKey, credentialID, counter } =
      verification.registrationInfo;

    await db.query(
      `INSERT INTO user_credentials 
       (id, user_id, credential_id, public_key, sign_count)
       VALUES ($1,$2,$3,$4,$5)`,
      [uuid(), req.user.id, credentialID, credentialPublicKey, counter]
    );

    res.json({ success: true });
  } else {
    res.status(400).json({ error: "Verification failed" });
  }
});

  const options = generateAuthenticationOptions({
    allowCredentials: credentials.map(c => ({
      id: c.credential_id,
      type: "public-key"

  await redis.set(`challenge:${user.id}`, options.challenge, "EX", 300);
  res.json(options);
});
🔎 Verify Authentication + Replay Protection
import { verifyAuthenticationResponse } from "@simplewebauthn/server";

app.post("/auth/webauthn/login/verify", async (req, res) => {
  const expectedChallenge = await redis.get(`challenge:${req.user.id}`);
  const credential = await getCredential(req.body.id);

  const verification = await verifyAuthenticationResponse({
    response: req.body,
    expectedChallenge,
    expectedOrigin: "https://fintekpro.com",
    expectedRPID: "fintekpro.com",
    authenticator: {
      credentialID: credential.credential_id,
      credentialPublicKey: credential.public_key,
      counter: credential.sign_count
    }
  });

  if (verification.verified) {
    if (verification.authenticationInfo.newCounter <= credential.sign_count) {
      throw new Error("Replay attack detected");
    }

    await updateSignCount(credential.id, verification.authenticationInfo.newCounter);

    res.json({ success: true });
  }
});
🔐 Cryptographic Flow Diagram
User → Browser → FintekPro Backend

1. User taps "Enable Face ID"
2. Backend generates challenge (random 32B)
3. Browser calls navigator.credentials.create()
4. Device signs challenge with private key (stored in Secure Enclave)
5. Public key + credentialID sent to backend
6. Backend verifies signature
7. Public key stored in DB

Authentication:

1. Backend sends new challenge
2. Device signs challenge
3. Backend verifies signature using stored public key
4. Counter validated (anti-replay)
5. Session issued (JWT)
Security Model:
Private key never leaves device
Only public key stored server-side
Cryptographic signature = proof of possession

Risk Calculation
function calculateRiskScore(context) {
  let score = 0;

  if (context.newDevice) score += 30;
  if (context.geoMismatch) score += 25;
  if (context.txnAmount > 500000) score += 40;
  if (context.inactiveDays > 30) score += 20;
  if (context.failedAttempts > 3) score += 50;

  return score;
}
Enforcement Middleware
function stepUpMiddleware(req, res, next) {
  const riskScore = calculateRiskScore(req.context);

  if (riskScore < 40) return next();
  if (riskScore < 70) return requireBiometric(req, res, next);
  return requireBiometricAndOTP(req, res, next);
}
