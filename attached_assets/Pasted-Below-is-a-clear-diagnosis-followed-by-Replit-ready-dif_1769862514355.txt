export function aggregateHolding(holding: Holding) {
  const totalUnits = holding.lots.reduce((s,l)=>s+l.units,0)
  const invested = holding.lots.reduce((s,l)=>s+l.amount,0)

  return {
    ...holding,
    totalUnits,
    investedValue: invested,
    avgNav: invested / totalUnits
  }
}
