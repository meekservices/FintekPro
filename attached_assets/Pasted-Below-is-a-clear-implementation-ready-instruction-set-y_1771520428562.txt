STEP 4 — Build Category Normalization Utility
Create:
/utils/normalizeCategory.ts
export function normalizeCategory(raw: string): string {
  if (!raw) return '';

  const value = raw.toLowerCase();

  if (value.includes('large') && value.includes('mid')) return 'large_mid';
  if (value.includes('large')) return 'large_cap';
  if (value.includes('mid')) return 'mid_cap';
  if (value.includes('small')) return 'small_cap';
  if (value.includes('flexi')) return 'flexi_cap';
  if (value.includes('elss')) return 'elss';
  if (value.includes('bank')) return 'banking';

  return 'unknown';
}

STEP 6 — Build Auto-Mapping Service
Create:
/services/benchmark/autoMapper.ts
Logic:
async function autoMapBenchmarks() {
  const funds = await db.query(`
    SELECT id, category 
    FROM mutual_funds
    WHERE id NOT IN (
      SELECT fund_id FROM fund_benchmark_mapping
    )
  `);

  let mapped = 0;
  let skipped = 0;

  for (const fund of funds.rows) {
    const normalized = normalizeCategory(fund.category);
    const benchmarkSymbol = categoryBenchmarkMap[normalized];

    if (!benchmarkSymbol) {
      skipped++;
      continue;
    }

    const benchmark = await db.query(
      `SELECT id FROM benchmarks WHERE symbol = $1`,
      [benchmarkSymbol]
    );

    if (!benchmark.rows.length) {
      skipped++;
      continue;
    }

    await db.query(`
      INSERT INTO fund_benchmark_mapping 
      (fund_id, benchmark_id, mapping_type, confidence_score)
      VALUES ($1, $2, 'category', 0.7)
      ON CONFLICT (fund_id) DO NOTHING
    `, [fund.id, benchmark.rows[0].id]);

    mapped++;
  }

  console.log(`Auto-mapped: ${mapped}, skipped: ${skipped}`);
}
