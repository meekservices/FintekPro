parseSchemeTransactions(schemeText: string): PortfolioLot[] {
  const lots: PortfolioLot[] = [];

  const transactionRegex =
    /^(\d{2}-[A-Za-z]{3}-\d{4})\s+
     (Purchase|SIP Purchase|Systematic Investment Purchase|Switch In)[^\d]*
     ([\d,]+\.\d{2})\s+
     ([\d,]+\.\d{3})\s+
     ([\d.]+)/gm;

  let match;
  while ((match = transactionRegex.exec(schemeText)) !== null) {
    lots.push({
      transactionDate: parseDate(match[1]),
      transactionType: normalizeType(match[2]),
      amount: parseNumber(match[3]),
      units: parseNumber(match[4]),
      nav: parseNumber(match[5]),
    });
  }

  return lots;
}
